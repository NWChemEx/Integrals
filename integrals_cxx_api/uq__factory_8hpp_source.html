48inline UQKind uq_kind_from_string(const std::string& uq_type) {
49 if(uq_type == "uncertain") return UQKind::uncertain;
50 if(uq_type == "interval") return UQKind::interval;
51 if(uq_type == "affine") return UQKind::affine;
52 if(uq_type == "thresholded affine") return UQKind::thresholded_affine;
53 if(uq_type == "taylor model") return UQKind::taylor_model;
54 throw std::runtime_error(
55 "integrals::property_types::uq_kind_from_string: Invalid UQ type name " +
56 uq_type);
57}

60inline std::string to_string(UQKind kind) {
61 switch(kind) {
62 case UQKind::uncertain: return "uncertain";
63 case UQKind::interval: return "interval";
64 case UQKind::affine: return "affine";

66 case UQKind::taylor_model: return "taylor model";
67 }
68 throw std::runtime_error(
69 "integrals::property_types::to_string(UQKind): Unrecognized UQKind");
70}

158template<typename T>
159T make_taylor_model(double center, double radius, std::size_t order) {
160 if constexpr(tensorwrapper::types::is_taylor_model_v<T>) {
161 return T(center - radius, center + radius, typename T::Order(order));
162 } else {
163 return tensorwrapper::types::construct_uq_type<T>(center, radius);
164 }

172class TaylorModelFactoryImpl final : public UQFactoryBase {
173public:
175 explicit TaylorModelFactoryImpl(std::size_t order = 2) : m_order_(order) {}
176
178 wtf::fp::Float operator()(double center, double radius) const override {
179 using uq_t = tensorwrapper::types::taylor_model_type<double>;
180 return wtf::fp::Float(
181 make_taylor_model<uq_t>(center, radius, m_order_));
182 }
183
185 UQKind kind() const noexcept override { return UQKind::taylor_model; }
186
188 std::size_t order() const noexcept { return m_order_; }
189
190 // Deep-copy *this
191 std::unique_ptr<UQFactoryBase> clone() const override {
192 return std::make_unique<TaylorModelFactoryImpl>(*this);
193 }
194
196 bool equal(const UQFactoryBase& rhs) const noexcept override {
197 if(rhs.kind() != UQKind::taylor_model) return false;
198 return m_order_ ==
199 static_cast<const TaylorModelFactoryImpl&>(rhs).m_order_;
200 }
201
202 bool less(const UQFactoryBase& rhs) const noexcept override {
203 if(kind() != rhs.kind()) return kind() < rhs.kind();
204 return m_order_ <
205 static_cast<const TaylorModelFactoryImpl&>(rhs).m_order_;
206 }
207
208private:
209 std::size_t m_order_;
210};

224class UQFactory {
225public:
228 UQFactory() : m_impl_(std::make_shared<detail::UncertainFactoryImpl>()) {}
229
232 explicit UQFactory(UQKind kind, std::size_t order = 2) :
233 m_impl_(make_impl_(kind, order)) {}
234
236 wtf::fp::Float operator()(double center, double radius) const {
237 return (*m_impl_)(center, radius);
238 }
239
241 UQKind kind() const noexcept { return m_impl_->kind(); }
242
244 bool operator==(const UQFactory& rhs) const noexcept {
245 return m_impl_->equal(*rhs.m_impl_);
246 }
247
249 bool operator<(const UQFactory& rhs) const noexcept {
250 return m_impl_->less(*rhs.m_impl_);
251 }
252
253private:
254 static std::shared_ptr<const UQFactoryBase> make_impl_(UQKind kind,
255 std::size_t order) {
256 switch(kind) {
257 case UQKind::uncertain:
258 return std::make_shared<detail::UncertainFactoryImpl>();
259 case UQKind::interval:
260 return std::make_shared<detail::IntervalFactoryImpl>();
261 case UQKind::affine:
262 return std::make_shared<detail::AffineFactoryImpl>();
263 case UQKind::thresholded_affine:
264 return std::make_shared<detail::ThresholdedAffineFactoryImpl>();
265 case UQKind::taylor_model:
266 return std::make_shared<detail::TaylorModelFactoryImpl>(order);
267 }
268 throw std::runtime_error(
269 "integrals::property_types::UQFactory: Unrecognized UQKind");
270 }
271
272 std::shared_ptr<const UQFactoryBase> m_impl_;
273};
